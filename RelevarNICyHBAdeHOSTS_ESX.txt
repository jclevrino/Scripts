### Leer los hosts ESXi en los que se quiere hacer el relevamiento

### directamente en el script se descomentará la linea 2
# 1
# $esxhosts = get-content C:\scripts\hostname.txt ## Leer un archivo con los HOST a relevar
# 2
# $esxhosts = "dc1_ESX01.dominio.int","dc1_ESX02.dominio.int" ## Listar manualmente los hosts

foreach ($esx in $esxhosts)
{

  $esxcli = Get-EsxCli -VMHost $esx

  $esxcli.Storage.san.fc.list.Invoke()|select @{N='HostName';E={$esxcli.VMHost.Name}}, Adapter,DriverName,DriverVersion,FirmwareVersion,hardwareVersion,PortState # |export-csv -path C:\Scripts\HBA-DriverFW.csv

  $Nics =$esxcli.network.nic.list()

  foreach ($nic in $Nics)
     {

      $Nicinfo = $Esxcli.network.nic.get($nic.name).driverinfo

      $Nicinfo | Select @{N="HostName";E={$esxcli.VMHost.Name}},

                                @{N="NICName";E={$nic.Name}},

                                @{N="NICDriver";E={$NicInfo.Driver}},

                                @{n="NICDriver-Version";E={$NicInfo.Version}},

                                @{N="NICFirmware-Version";E={$NicInfo.FirmWareVersion}} ### si se desea salida en un archivo borrar este comentario |Export-Csv -Path C:\Scripts\NIC-Drive-FW.csv

     }
